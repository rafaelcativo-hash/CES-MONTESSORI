async function aplicarRestriccionMateriaDocente() { const selectMateria = document.getElementById('cal-materia'); if (!selectMateria) return; // Si es admin, mostrar todas las opciones completas sin restricciones if (usuarioRolActual === 'admin') { for (let opt of selectMateria.options) { opt.disabled = false; opt.style.display = ''; } return; } // Normalizar la especialidad del docente para hacer una búsqueda flexible const esp = docenteEspecialidadGlobal.toLowerCase(); for (let opt of selectMateria.options) { if (!opt.value) continue; const valMateria = opt.value.toLowerCase(); let mostrarOpcion = false; // Lógica de correspondencia estricta por especialidad docente if (esp.includes('percusión') || esp.includes('percursion')) { if (valMateria.includes('percusión') || valMateria.includes('percursion')) mostrarOpcion = true; } else if (esp.includes('canto')) { if (valMateria.includes('canto')) mostrarOpcion = true; } else if (esp.includes('danza')) { if (valMateria.includes('danza')) mostrarOpcion = true; } else if (esp.includes('plástica') || esp.includes('plastica')) { if (valMateria.includes('plásticas') || valMateria.includes('plastica')) mostrarOpcion = true; } else if (esp.includes('solfeo')) { if (valMateria.includes('solfeo')) mostrarOpcion = true; } else if (esp.includes('piano')) { if (valMateria.includes('piano')) mostrarOpcion = true; } else if (esp.includes('guitarra')) { if (valMateria.includes('guitarra')) mostrarOpcion = true; } else if (esp.includes('ukulele')) { if (valMateria.includes('ukulele')) mostrarOpcion = true; } else if (esp.includes('batería') || esp.includes('bateria')) { if (valMateria.includes('batería') || valMateria.includes('bateria')) mostrarOpcion = true; } else if (esp.includes('bajo')) { if (valMateria.includes('bajo')) mostrarOpcion = true; } else if (esp.includes('inglés') || esp.includes('ingles')) { if (valMateria.includes('inglés') || valMateria.includes('ingles')) mostrarOpcion = true; } else if (esp.includes('edufi') || esp.includes('educación física')) { if (valMateria.includes('edufi')) mostrarOpcion = true; } else if (esp.includes('académico') || esp.includes('academico') || esp.includes('español') || esp.includes('matemáticas')) { // Los docentes académicos ven Español, Matemáticas, Ciencias, Estudios Sociales y Conducta if (valMateria.includes('español') || valMateria.includes('matemáticas') || valMateria.includes('ciencias') || valMateria.includes('estudios sociales') || valMateria.includes('conducta')) { mostrarOpcion = true; } } // Aplicar estado visual y bloqueo en el select if (mostrarOpcion) { opt.disabled = false; opt.style.display = ''; } else { opt.disabled = true; opt.style.display = 'none'; } } // Seleccionar automáticamente la primera opción disponible para el docente let seleccionada = false; for (let opt of selectMateria.options) { if (!opt.disabled && opt.value) { selectMateria.value = opt.value; seleccionada = true; break; } } if (!seleccionada) { selectMateria.value = ''; } cargarGrupoParaCalificar(); }
